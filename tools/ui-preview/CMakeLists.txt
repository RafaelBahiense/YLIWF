cmake_minimum_required(VERSION 3.25)
project(YLIWFPreview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
get_filename_component(REPO_ROOT "../.." ABSOLUTE)
find_package(imgui CONFIG REQUIRED)
find_package(fmt CONFIG REQUIRED)
file(READ "${REPO_ROOT}/mod.json" identity)
foreach(pair "displayName;DisplayName" "pluginFile;PluginFile" "binaryName;BinaryName")
  list(GET pair 0 key)
  list(GET pair 1 name)
  string(JSON value GET "${identity}" "${key}")
  string(APPEND preview_identity "inline constexpr char ${name}[] = \"${value}\";\n")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/ModInfo.h" CONTENT
  "#pragma once\nnamespace mod::info {\n${preview_identity}inline constexpr char IniPath[] = \"Preview settings (memory only)\";\n}\n")
include(Theme.cmake)
add_library(yliwf-preview-ui SHARED UIModule.cpp Simulation.cpp Style.cpp
  "${REPO_ROOT}/src/native/ui/UI.cpp"
  "${REPO_ROOT}/src/native/ui/HotkeysPage.cpp")
add_custom_command(TARGET yliwf-preview-ui POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E touch "${CMAKE_CURRENT_BINARY_DIR}/ui-ready")
add_executable(yliwf-ui-preview WIN32 Main.cpp ReloadableUI.cpp)
set_target_properties(yliwf-ui-preview PROPERTIES OUTPUT_NAME yliwf-ui-preview-host)
add_dependencies(yliwf-ui-preview yliwf-preview-ui)
foreach(target yliwf-preview-ui yliwf-ui-preview)
  target_include_directories(${target} PRIVATE include "${REPO_ROOT}/src/native"
    "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_precompile_headers(${target} PRIVATE <algorithm> <array> <atomic> <mutex> <string> <vector>)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  target_link_libraries(${target} PRIVATE fmt::fmt d3d11 dxgi)
endforeach()
target_link_libraries(yliwf-preview-ui PRIVATE imgui::imgui)
