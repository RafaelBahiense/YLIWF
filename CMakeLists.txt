cmake_minimum_required(VERSION 3.25)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/mod.json" MOD_IDENTITY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/mod.json")
function(read_identity key variable)
  string(JSON value ERROR_VARIABLE error GET "${MOD_IDENTITY}" "${key}")
  if(error OR value STREQUAL "" OR value MATCHES "[\r\n\t]")
    message(FATAL_ERROR "mod.json: missing or invalid ${key}")
  endif()
  set(${variable} "${value}" PARENT_SCOPE)
endfunction()
read_identity(displayName MOD_DISPLAY_NAME)
read_identity(shortName MOD_SHORT_NAME)
read_identity(pluginFile MOD_PLUGIN_FILE)
read_identity(binaryName MOD_BINARY_NAME)
read_identity(scriptPrefix MOD_SCRIPT_PREFIX)
read_identity(author AUTHOR_NAME)
foreach(value MOD_SHORT_NAME MOD_BINARY_NAME)
  if(NOT "${${value}}" MATCHES "^[A-Za-z_][A-Za-z0-9_-]*$")
    message(FATAL_ERROR "mod.json: ${value} must be an identifier")
  endif()
endforeach()
if(NOT MOD_SCRIPT_PREFIX MATCHES "^[A-Za-z_][A-Za-z0-9_]*$" OR
   NOT MOD_PLUGIN_FILE MATCHES "^[^<>:\"/\\\\|?*;]+\\.[Ee][Ss][Pp]$")
  message(FATAL_ERROR "mod.json: invalid script prefix or plugin filename")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/papyrus/core/${MOD_SCRIPT_PREFIX}_SKSE.psc")
  message(FATAL_ERROR "mod.json: scriptPrefix requires matching Papyrus sources and native bindings")
endif()
foreach(value MOD_DISPLAY_NAME AUTHOR_NAME)
  string(REPLACE "\\" "\\\\" ${value}_CPP "${${value}}")
  string(REPLACE "\"" "\\\"" ${value}_CPP "${${value}_CPP}")
  string(REPLACE "\n" "\\n" ${value}_CPP "${${value}_CPP}")
  string(REPLACE "\r" "\\r" ${value}_CPP "${${value}_CPP}")
endforeach()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" MOD_VERSION LIMIT_COUNT 1)
project(Mod VERSION ${MOD_VERSION} LANGUAGES CXX)
if("${PROJECT_VERSION_TWEAK}" STREQUAL "")
  set(PROJECT_VERSION_TWEAK 0)
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(COMMONLIB_SSE_FOLDER "$ENV{COMMONLIB_SSE_FOLDER}" CACHE PATH "CommonLib source")
if(NOT COMMONLIB_SSE_FOLDER)
  set(COMMONLIB_SSE_FOLDER "${CMAKE_CURRENT_SOURCE_DIR}/.tools/CommonLibSSE-NG" CACHE PATH "" FORCE)
endif()
if(NOT EXISTS "${COMMONLIB_SSE_FOLDER}/CMakeLists.txt")
  message(FATAL_ERROR "Run setup.ps1 or set COMMONLIB_SSE_FOLDER")
endif()
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "Build CommonLib tests" FORCE)
# Keep local paths for development debugging; published Release binaries use
# stable paths, including source_location strings compiled into CommonLib.
file(TO_NATIVE_PATH "${CMAKE_CURRENT_SOURCE_DIR}" RELEASE_SOURCE_PATH)
file(TO_NATIVE_PATH "${COMMONLIB_SSE_FOLDER}" RELEASE_COMMONLIB_PATH)
add_compile_options(
  "$<$<CONFIG:Release>:/experimental:deterministic>"
  "$<$<CONFIG:Release>:/pathmap:${RELEASE_SOURCE_PATH}=source>"
  "$<$<CONFIG:Release>:/pathmap:${RELEASE_COMMONLIB_PATH}=dependencies\\CommonLibSSE-NG>")
add_subdirectory("${COMMONLIB_SSE_FOLDER}" "${CMAKE_CURRENT_BINARY_DIR}/_deps/clib-build")
find_path(SKSE_MCP_INCLUDE_DIRS "SKSEMCP/utils.hpp" REQUIRED)
configure_file(cmake/version.rc.in version.rc @ONLY)
configure_file(cmake/ModInfo.h.in generated/ModInfo.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/settings.ini" MOD_DEFAULT_INI)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/settings.ini")
configure_file(cmake/SettingsDefaults.h.in generated/SettingsDefaults.h @ONLY)
# configure_file preserves the timestamp when plugin metadata is unchanged.
# CommonLib's add_commonlibsse_plugin rewrites its generated .cpp every configure.
set(PLUGIN_BINARY_NAME "${MOD_BINARY_NAME}")
configure_file(cmake/plugin_info.cpp.in plugin_info.cpp @ONLY)
# Keep discovery scoped to the core; add-ons and tests own separate targets.
file(GLOB_RECURSE MOD_SOURCES CONFIGURE_DEPENDS
  "${PROJECT_SOURCE_DIR}/src/native/*.cpp"
  "${PROJECT_SOURCE_DIR}/include/*.h" "${PROJECT_SOURCE_DIR}/include/*.hpp")
add_library(mod SHARED ${MOD_SOURCES}
  "${CMAKE_CURRENT_BINARY_DIR}/version.rc" "${CMAKE_CURRENT_BINARY_DIR}/plugin_info.cpp")
include(cmake/NativePlugins.cmake)
configure_native_plugin(mod "${MOD_BINARY_NAME}")
target_precompile_headers(mod PRIVATE include/PCH.h)
add_custom_target(native_plugins DEPENDS mod)
enable_testing()
add_custom_target(native_tests)
foreach(test debug_rules controller_rules adapter_rules)
  add_native_test(${test} "${PROJECT_SOURCE_DIR}/tests/native/${test}.cpp")
endforeach()
add_native_addons()

find_package(fmt CONFIG REQUIRED)
target_link_libraries(debug_rules_test PRIVATE fmt::fmt)
target_include_directories(debug_rules_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(debug_rules_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
